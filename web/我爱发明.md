
写脚本高并发就行
```sh
#!/usr/bin/env bash

STAGE_URL="http://127.0.0.1:12345/stage/3"
INVENT_BASE="http://127.0.0.1:12345/invent"
FINALIZE_URL="http://127.0.0.1:12345/api/finalize"

# ========== Step 1: GET /stage/3，取 race cookie ==========
HEADERS=$(curl -D - $STAGE_URL)

RACE=$(echo "$HEADERS" \
  | grep -i '^Set-Cookie:' \
  | grep -o 'race=[^;]*' \
  | head -1 \
  | cut -d= -f2-)

if [[ -z "$RACE" ]]; then
  echo "❌ 未能从 /stage/3 读取 race cookie" >&2
  exit 1
fi

echo "[*] race=${RACE}" >&2

# ========== Step 2: 并发请求 /invent/1..288 ==========
# 临时目录存放每个 ID 的结果
TMPDIR=$(mktemp -d)
trap 'rm -rf "$TMPDIR"' EXIT

export INVENT_BASE RACE TMPDIR

# 每个 ID 的请求逻辑（作为 xargs 子进程）
fetch_one() {
  local ID="$1"
  local HTML
  HTML=$(curl -s --max-time 5 \
    -H "Cookie: race=${RACE}" \
    "${INVENT_BASE}/${ID}")

  if [[ $? -ne 0 || -z "$HTML" ]]; then
    return
  fi

  local CODE
  CODE=$(echo "$HTML" \
    | sed -n 's/.*<code class="inspection-code"[^>]*data-code="\([^"]*\)".*/\1/p')

  if [[ -n "$CODE" ]]; then
    # 输出 "ID|CODE" 到临时文件
    echo "${ID}|${CODE}" > "${TMPDIR}/${ID}"
  fi
}

export -f fetch_one

# 并发 32 路（可根据机器调整）
seq 1 288 | xargs -P 32 -I {} bash -c 'fetch_one "$@"' _ {}

echo "[*] 并发请求完成，收集结果..." >&2

# ========== Step 3: 按 ID 排序，拼接 ==========
RESULT=""
FOUND=0

for ID in $(ls "$TMPDIR" | sort -n); do
  CODE=$(cat "${TMPDIR}/${ID}" | cut -d'|' -f2)
  if [[ $FOUND -eq 0 ]]; then
    RESULT="$CODE"
    FOUND=1
  else
    RESULT="${RESULT}|${CODE}"
  fi
done

if [[ $FOUND -eq 0 ]]; then
  echo "❌ 未提取到任何 inspection-code" >&2
  exit 2
fi

# ========== Step 4: SHA256 ==========
SHA256=$(printf '%s' "$RESULT" | sha256sum | awk '{print $1}')
echo "[*] SHA256: $SHA256" >&2

# ========== Step 5: POST /api/finalize ==========
RESPONSE=$(curl -s -X POST "$FINALIZE_URL" \
  -H "Content-Type: application/json" \
  -H "Cookie: race=${RACE}" \
  -d "{\"answer\":\"${SHA256}\"}")

echo "[*] 服务器响应:" >&2
echo "$RESPONSE" >&2

# 最终只输出 SHA256
echo "$SHA256"
```
